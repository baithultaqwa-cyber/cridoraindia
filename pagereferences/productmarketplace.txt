                    <span className="text-slate-400">GST on Gold (3%)</span>
                    <span>₹{p.gstOnGold.toLocaleString(undefined, {maximumFractionDigits: 2})}</span>
                  </div>
                  <div className="flex justify-between text-sm">
                    <span className="text-slate-400">Making Charges</span>
                    <div className="text-right">
                      <span className="line-through text-slate-500 mr-2 text-xs">₹{(p.makingCharges + p.discountAmount).toLocaleString()}</span>
                      <span>₹{p.makingCharges.toLocaleString()}</span>
                    </div>

                    <span className="text-slate-400">GST on Making (18%)</span>

                    <span>Vault Discount</span>
                    <span>- ₹{p.discountAmount.toLocaleString()}</span>
                  </div>
                </div>

                  <div>
                    <p className="text-slate-400 text-xs font-black uppercase tracking-widest mb-1">Order Total</p>
                    <p className="text-3xl font-black">₹{p.finalAmount.toLocaleString(undefined, {maximumFractionDigits: 0})}</p>

                    Prices secured for 04:59 mins

                <h3 className="text-xl font-black mb-8">Payment Strategy</h3>
                
                <div className="space-y-4">
                  {/* Vault Payment Option */}

                        <div className="w-10 h-10 bg-amber-500 rounded-xl flex items-center justify-center text-slate-950">
                          <Wallet className="w-5 h-5" />

                      <div className="flex justify-between">
                        <span className="text-slate-400">Redeeming</span>
                        <span className="font-bold text-white">{p.goldFromVault.toFixed(3)}g</span>
                      </div>
                      <div className="flex justify-between">
                        <span className="text-slate-400">Vault Credit</span>
                        <span className="font-bold text-green-400">₹{p.vaultValueOffset.toLocaleString(undefined, {maximumFractionDigits: 0})}</span>
                      </div>
                    </div>
                  </div>

                  {/* Dynamic Pay Button */}
                  {isVaultOnly ? (
                    <button className="w-full bg-amber-500 text-slate-950 py-6 rounded-[24px] font-black text-xl hover:bg-amber-400 transition-all flex items-center justify-center gap-3 shadow-2xl shadow-amber-500/20">

        <div className="absolute top-0 right-0 p-8 z-10 flex items-center gap-6">
           <div className="text-right hidden md:block">
              <p className="text-[10px] font-black uppercase tracking-widest text-amber-500">Global Vault Balance</p>
              <p className="text-2xl font-black text-white">{userVaultBalance.toFixed(3)}g</p>
           </div>
           <div className="w-14 h-14 bg-white/5 backdrop-blur-2xl rounded-2xl flex items-center justify-center border border-white/10 hover:bg-white/10 transition-colors cursor-pointer relative">
              <ShoppingBag className="w-6 h-6 text-amber-500" />
              <span className="absolute -top-1 -right-1 w-4 h-4 bg-amber-500 text-slate-950 text-[10px] font-bold rounded-full flex items-center justify-center">0</span>
           </div>
        </div>

        <div className="relative z-10 max-w-7xl w-full mx-auto">
          <div className="flex items-center gap-3 mb-4">
             <span className="h-px w-12 bg-amber-500"></span>
             <span className="text-[10px] font-black uppercase tracking-[0.3em] text-amber-500">Member Catalog Q2 2026</span>
          </div>
          <h1 className="text-7xl md:text-9xl font-black tracking-tighter mb-8 uppercase italic leading-[0.8]">Jewel<br/><span className="text-transparent border-text-white" style={{WebkitTextStroke: '1px white'}}>Vault</span></h1>
          <div className="flex flex-wrap gap-4 items-center">
            <div className="relative flex-1 max-w-xl group">

            return (

                key={p.id} 
                className="group bg-gradient-to-b from-white/10 to-transparent border border-white/10 rounded-[40px] p-5 hover:border-amber-500/50 transition-all duration-500 flex flex-col relative overflow-hidden"
              >
                <div className="relative aspect-[1/1] rounded-[32px] overflow-hidden mb-6 cursor-pointer" onClick={() => setSelectedProduct(p)}>

                  <div className="absolute inset-0 bg-gradient-to-t from-slate-950/80 via-transparent to-transparent opacity-60"></div>
                  
                  <div className="absolute top-4 left-4 flex gap-2">
                    {p.isXRedeem && <span className="bg-amber-500 text-slate-950 px-2 py-1 rounded-lg text-[8px] font-black uppercase">X-Redeem</span>}
                  </div>

                  <button className="absolute bottom-4 right-4 w-12 h-12 bg-white/10 backdrop-blur-md rounded-full flex items-center justify-center text-white opacity-0 group-hover:opacity-100 translate-y-4 group-hover:translate-y-0 transition-all duration-300 hover:bg-amber-500 hover:text-slate-950 shadow-2xl">
                    <Plus className="w-6 h-6" />
                  </button>
                </div>

                <div className="flex-1 px-1 cursor-pointer" onClick={() => setSelectedProduct(p)}>

                    <div>
                       <p className="text-[9px] font-black text-slate-500 uppercase tracking-widest mb-1">{p.jeweller}</p>
                       <h3 className="font-bold text-lg leading-tight group-hover:text-amber-400 transition-colors line-clamp-1">{p.name}</h3>

                  <div className="flex items-center gap-4 my-4">
                    <div className="flex items-center gap-1.5 text-xs text-slate-300 font-bold bg-white/5 px-2 py-1 rounded-lg">

                <ShieldCheck className="w-6 h-6 text-blue-500" /> Price Estimate

                    <div className="space-y-4">
                      <div className="flex justify-between items-center text-sm">
                        <span className="text-slate-400 font-bold">Gold Value</span>
